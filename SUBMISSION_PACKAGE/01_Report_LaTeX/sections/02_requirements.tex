\section{Yêu cầu chức năng theo vai trò}

\subsection{Khách hàng}

\begin{itemize}[leftmargin=1.2cm]
    \item Đăng ký tài khoản bằng email hoặc số điện thoại, có kiểm tra trùng.
    \item Đăng nhập, quên mật khẩu, đổi mật khẩu và cập nhật hồ sơ cá nhân.
    \item Tìm kiếm voucher theo từ khóa và lọc theo danh mục, khu vực, giá, mức giảm, đối tác.
    \item Xem chi tiết voucher gồm giá gốc, giá bán, thời hạn, điều kiện, chi nhánh và đánh giá.
    \item Quản lý giỏ hàng ở phía client: thêm, cập nhật số lượng, xóa voucher và xem tổng tiền tạm tính.
\end{itemize}

\subsection{Đối tác}

\begin{itemize}[leftmargin=1.2cm]
    \item Đăng ký tài khoản đối tác cùng thông tin doanh nghiệp, người đại diện, mã số thuế, trụ sở và chi nhánh.
    \item Tạo voucher với danh mục, giá, mô tả, hình ảnh, thời gian bán, thời hạn sử dụng, chính sách hoàn/hủy và chi nhánh áp dụng.
    \item Gửi voucher sang trạng thái chờ duyệt và theo dõi trạng thái.
    \item Quản lý voucher trong phạm vi cho phép, xem số lượng phát hành, đã bán, đã dùng.
    \item Kiểm tra voucher code và xác nhận sử dụng tại chi nhánh hợp lệ.
    \item Xem dashboard/báo cáo doanh thu, số lượng bán, tỷ lệ sử dụng và hiệu quả từng chương trình.
\end{itemize}

\subsection{Quản trị viên}

\begin{itemize}[leftmargin=1.2cm]
    \item Xem danh sách đối tác chờ duyệt, duyệt hoặc từ chối hồ sơ.
    \item Xem, tìm kiếm, lọc, khóa/mở khóa tài khoản người dùng và đổi vai trò.
    \item Xem danh sách voucher theo trạng thái: Pending, Approved, Rejected, Suspended.
    \item Duyệt voucher, từ chối kèm lý do hoặc tạm ẩn voucher.
    \item Xem dashboard tổng quan về người dùng, đối tác, voucher và các chỉ số mô phỏng.
\end{itemize}

\section{Quy tắc nghiệp vụ}

\begin{itemize}[leftmargin=1.2cm]
    \item Voucher chỉ được bán khi đã được quản trị viên duyệt.
    \item Giá bán voucher phải nhỏ hơn giá gốc.
    \item Voucher phải có thời gian bắt đầu và hết hạn hợp lệ.
    \item Số lượng tồn không âm và không vượt quá tổng số lượng phát hành.
    \item Voucher code là duy nhất và không được sử dụng lại sau khi đã chuyển sang trạng thái Used.
    \item Đối tác chỉ được xác thực voucher thuộc chương trình hoặc chi nhánh của mình.
    \item Người dùng chỉ được đánh giá voucher sau khi có giao dịch liên quan.
\end{itemize}
